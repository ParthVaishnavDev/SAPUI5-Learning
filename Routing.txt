1. Create XML views

Home.view.xml
Detail.view.xml

2. Create controllers

Home.controller.js
Detail.controller.js

3. Link XML → controller

controllerName="your.app.controller.Home"

4. Add button press event in XML

<Button text="Go" press="onGoToDetail"/>

5. Write navTo() in controller

onGoToDetail: function () {
    this.getOwnerComponent().getRouter().navTo("Detail");
}

6. Define routes + targets in manifest.json

"routes": [
  {
    "pattern": "",
    "name": "Home",
    "target": "Home"
  },
],

"targets": {
  "Home": {
    "viewName": "Home"
  },
}


7. Ensure App container exists

<App id="app"/>

8. Initialize router in Component.js

"controlId": "app"
